function donkeyGetHay() {
  return new Promise((resolve, reject) => {
    console.log("Donkey hay lene gaya... 🐴");
    
    setTimeout(() => {
      const hayMili = Math.random() > 0.3;  // 70% chance mili
      
      if (hayMili) {
        resolve("Hay mil gayi 🌾");   // ✅
      } else {
        reject("Donkey bhaag gaya 💥"); // ❌
      }
    }, 2000);
  });
}
