//now here is an example of dockerfile for a backend java script project-
#the version of NODE, this is the start of the dockerfile
FROM node:22

#go to the directory, just like cd in terminal
WORKDIR /app

#copy the package json file(if available)
COPY package*.json ./

#Install all dependencies
RUN npm install

#copy the app into the container after the package.json
COPY ..

#set port environment variable
ENV PORT=9000

#exposing it so computer can excess it
EXPOSE 9000

#run the app
CMD ["npm","start"]

>docker run -p 9000:9000 <name of the image>

the left 9000 is computer port and the right one is container port
